MainNavigation : BaseNavigationC#
// 1. 页面登记表：ViewType 即 GoTo 的 key
public sealed class MainNavigation
    : BaseNavigation
{
    public MainNavigation(IGetService svc)
        : base(svc) { }

    public override string NavigationName
        => "Main";

    public override List<NavigateModel> Navigates
        => new()
    {
        new NavigateModel
        {
            ViewType = "OverviewView",
            Type     = typeof(OverviewView),
            VMType   = typeof(OverviewViewModel),
            Name     = "概览",
        },
        // …其余 7 页同构
    };

    public override void InitView() { }
}

// 2. 侧栏点击 → 一行调用
navigation.GoTo("ProgressView");

// 强类型参数：VM 实现 INavigateIn<T>
navigation.GoTo("DeviceView", "SN-0001");
三条约定
01
ViewType 是唯一 key

字符串匹配 Navigates，找不到返回 null，不会抛异常。

02
ViewModel 决定收不收参数

实现 INavigateIn 收无参调用，实现 INavigateIn<T> 收参数，两者都在 GoTo 里被自动识别。

03
离开即释放

旧视图的 DataContext 若实现 INavigateOut，先回调再换视图——轮询与订阅在这里收尾。

四项注意
密钥不要硬编码在客户端

至少做一次运行时派生，或由服务端下发短效密钥。

IV 必须随机

SetIV 传入固定 IV 等于退化为 ECB。

MD5 只做校验

摘要不等于加密，口令存储请用带盐的慢哈希。

指纹可伪造

硬件 ID 能改，授权方案需配合服务端校验。
